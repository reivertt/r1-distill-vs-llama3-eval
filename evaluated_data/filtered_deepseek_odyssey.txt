    for i in range(len(string), 0, -1):
        suffix = string[-i:]
        if is_palindrome(suffix):
            max_suffix = suffix
            max_suffix_length = i

def find_zero(xs: list):
    if len(xs) % 2 != 0:
        raise ValueError("xs must have even number of coefficients")
    if all(coeff == 0 for coeff in xs):
        raise ValueError("zero polynomial has no zero point")
    if len(xs) == 0:
        return 0.0
    n = len(xs)
    highest_degree = n - 1
    if highest_degree == 0:
        if xs[0] == 0:
            raise ValueError("constant polynomial must have a zero only if all coefficients are zero")
        return 0.0
    if highest_degree == 1:
        # f(x) = a + b x
        # f(x) = 0 => a + b x = 0 => x = -a / b
        if xs[0] == 0:
            raise ValueError("linear polynomial must have a zero only if all coefficients are zero")
        return -xs[0] / xs[1]
    n = len(xs)
    highest_degree = n - 1

    if highest_degree == 0:
        if xs[0] == 0:
            raise ValueError("constant polynomial must have a zero only if all coefficients are zero")
        return 0.0
    if highest_degree == 1:
        if xs[0] == 0:
            raise ValueError("linear polynomial must have a zero only if all coefficients are zero")
        return -xs[0] / xs[1]
    else:
        a_n = xs[-1]
        if a_n == 0:
            raise ValueError("zero polynomial has no zero point")
        degree = highest_degree
        if degree == 0:
            return 0.0
        if degree == 1:
            return -xs[0] / xs[1]
        else:
            # Compute x = (1 / a_n)^(1 / degree)
            # But this might not be the actual root, but it's a solution.
            x = (1.0 / a_n) ** (1.0 / degree)
            return x

def int_to_mini_roman(number):
    roman_numerals = [
        (1000, 'M'),
        (900, 'CM'),
        (500, 'D'),
        (400, 'CD'),
        (100, 'C'),
        (90, 'XC'),
        (50, 'L'),
        (40, 'XL'),
        (10, 'X'),
        (9, 'IX'),
        (5, 'V'),
        (4, 'IV'),
        (1, 'I')
    ]
    result = []
    for value, symbol in roman_numerals:
        while number >= value:
            result.append(symbol)
            number -= value
    return ''.join(result).lower()

def order_by_points(nums):
    # To sort the list based on sum of digits, then original index
    # First, create a list of tuples containing (sum of digits, original index, number)
    indexed_nums = [(num, i) for i, num in enumerate(nums)]
    # Then, create a helper function to compute sum of digits
    def sum_digits(x):
        return sum(int(d) for d in str(abs(x)))
    # Now, sort the list of tuples based on sum_digits and original index
    sorted_tuples = sorted(indexed_nums, key=lambda x: (sum_digits(x[0]), x[1]))
    # Extract the numbers in the new order
    result = [num for (num, i) in sorted_tuples]
    return result

def solve(s):
    processed = []
    for c in s:
        if c.isalpha():
            processed.append(c.swapcase())
        else:
            processed.append(c)
